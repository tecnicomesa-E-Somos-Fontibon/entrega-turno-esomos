☁️ Configurar Sincronización entre Equipos (GitHub) + Envío Gmail

Este código de Google Apps Script guarda el último estado permanentemente en Google Drive y nunca lo borra al abrir la página en otro computador:

  1. Abre https://script.google.com y asegúrate de tener este código con bloqueo anti-colisiones (LockService):
    var FILE_NAME = "esomos_entrega_turno_cloud_state.json"; function getOrCreateFile() { var files = DriveApp.getFilesByName(FILE_NAME); if (files.hasNext()) return files.next(); return DriveApp.createFile(FILE_NAME, "{}", MimeType.PLAIN_TEXT); } function doGet(e) { var file = getOrCreateFile(); var content = file.getBlob().getDataAsString() || "{}"; return ContentService.createTextOutput(content) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var lock = LockService.getScriptLock(); try { lock.waitLock(15000); var data = JSON.parse(e.postData.contents); // 1. Guardar estado sincronizado entre PC de oficina y portátil if (data.action === "save_state") { var file = getOrCreateFile(); file.setContent(JSON.stringify(data.state)); return ContentService.createTextOutput(JSON.stringify({ok: true, status: "saved", updatedAt: data.state.updatedAt})) .setMimeType(ContentService.MimeType.JSON); } // 2. Enviar el correo de entrega de turno directamente al jefe var inlineImages = {}; if (data.images) { for (var key in data.images) { var bytes = Utilities.base64Decode(data.images[key].base64); inlineImages[key] = Utilities.newBlob(bytes, data.images[key].mimeType, key); } } MailApp.sendEmail({ to: data.to, cc: data.cc || "", subject: data.subject, htmlBody: data.htmlBody, inlineImages: inlineImages }); return ContentService.createTextOutput(JSON.stringify({ok: true, status: "sent"})) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
  2. Recuerda que en Implementar → Gestionar implementaciones, en "Quién tiene acceso" debe estar en Cualquier persona (y que la URL sea la estándar https://script.google.com/macros/s/.../exec sin /a/macros/e-somos.co/).
